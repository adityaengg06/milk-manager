/* ========================================================= MILK MANAGER DATABASE SYSTEM ========================================================= */ /* ========================================================= DATABASE KEY ========================================================= */ const MILK_MANAGER_DB = "milkManagerDB"; /* ========================================================= GENERATE ID ========================================================= */ function generateID(prefix) { return ( prefix + Date.now() + Math.random() .toString(36) .substring(2, 7) ); } /* ========================================================= GET DATABASE ========================================================= */ function getDB() { const saved = localStorage.getItem( MILK_MANAGER_DB ); if (!saved) { return { customers: [], deliveries: [], bills: [], payments: [] }; } try { const db = JSON.parse(saved); /* Make sure all sections exist */ if (!Array.isArray(db.customers)) { db.customers = []; } if (!Array.isArray(db.deliveries)) { db.deliveries = []; } if (!Array.isArray(db.bills)) { db.bills = []; } if (!Array.isArray(db.payments)) { db.payments = []; } return db; } catch (error) { console.error( "Database error:", error ); return { customers: [], deliveries: [], bills: [], payments: [] }; } } /* ========================================================= SAVE DATABASE ========================================================= */ function saveDB(db) { try { localStorage.setItem( MILK_MANAGER_DB, JSON.stringify(db) ); return true; } catch (error) { console.error( "Unable to save database:", error ); alert( "Unable to save data." ); return false; } } /* ========================================================= GET CUSTOMERS ========================================================= */ function getCustomers() { const db = getDB(); return db.customers || []; }
