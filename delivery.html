/* ========================================================= MILK MANAGER DATABASE SYSTEM ========================================================= */ const MILK_MANAGER_DB = "milkManagerDB"; const OLD_CUSTOMERS_KEY = "milkManagerCustomers"; /* ========================================================= GENERATE ID ========================================================= */ function generateID(prefix) { return ( prefix + Date.now() + Math.random() .toString(36) .substring(2, 7) ); } /* ========================================================= GET DATABASE ========================================================= */ function getDB() { const saved = localStorage.getItem( MILK_MANAGER_DB ); let db; if (saved) { try { db = JSON.parse(saved); } catch (error) { console.error( "Database JSON error:", error ); db = null; } } if (!db) { db = { customers: [], deliveries: [], bills: [], payments: [] }; } /* Make sure sections exist */ if (!Array.isArray(db.customers)) { db.customers = []; } if (!Array.isArray(db.deliveries)) { db.deliveries = []; } if (!Array.isArray(db.bills)) { db.bills = []; } if (!Array.isArray(db.payments)) { db.payments = []; } /* IMPORT OLD CUSTOMERS if database has no customers */ if (db.customers.length === 0) { const oldCustomers = localStorage.getItem( OLD_CUSTOMERS_KEY ); if (oldCustomers) { try { const parsed = JSON.parse( oldCustomers ); if ( Array.isArray(parsed) && parsed.length > 0 ) { db.customers = parsed; localStorage.setItem( MILK_MANAGER_DB, JSON.stringify(db) ); } } catch (error) { console.error( "Old customer data error:", error ); } } } return db; } /* ========================================================= SAVE DATABASE ========================================================= */ function saveDB(db) { try { localStorage.setItem( MILK_MANAGER_DB, JSON.stringify(db) ); /* Keep old customer storage synchronized for compatibility. */ if ( Array.isArray(db.customers) ) { localStorage.setItem( OLD_CUSTOMERS_KEY, JSON.stringify( db.customers ) ); } return true; } catch (error) { console.error( "Unable to save database:", error ); alert( "Unable to save data." ); return false; } } /* ========================================================= GET CUSTOMERS ========================================================= */ function getCustomers() { const db = getDB(); return db.customers || []; }
