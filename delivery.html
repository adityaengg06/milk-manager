/* ========================================================= MILK MANAGER DELIVERY PAGE ========================================================= */ /* ========================================================= DELIVERY DATA ========================================================= */ let currentCustomerIndex = 0; /* ========================================================= GET CUSTOMERS ========================================================= */ function getCustomers() { const savedCustomers = localStorage.getItem("milkManagerCustomers"); if (!savedCustomers) { return []; } try { return JSON.parse(savedCustomers); } catch (error) { console.error( "Customer data error:", error ); return []; } } /* ========================================================= SAVE DELIVERY STATUS ========================================================= */ function saveDeliveryStatus( customerId, status, quantity, reason = "" ) { let deliveries = []; const savedDeliveries = localStorage.getItem( "milkManagerDeliveries" ); if (savedDeliveries) { try { deliveries = JSON.parse( savedDeliveries ); } catch (error) { deliveries = []; } } const today = new Date() .toISOString() .split("T")[0]; const existingIndex = deliveries.findIndex( function(delivery) { return ( delivery.customerId === customerId && delivery.date === today ); } ); const deliveryData = { customerId: customerId, date: today, status: status, quantity: quantity, reason: reason, updatedAt: new Date().toISOString() }; if (existingIndex !== -1) { deliveries[existingIndex] = deliveryData; } else { deliveries.push( deliveryData ); } localStorage.setItem( "milkManagerDeliveries", JSON.stringify( deliveries ) ); } /* ========================================================= MARK DELIVERED ========================================================= */ function markDelivered() { const customers = getCustomers(); if ( customers.length === 0 ) { alert( "No customers available." ); return; } const customer = customers[ currentCustomerIndex ]; if (!customer) { return; } const quantityElement = document.getElementById( "milkQty" ); let quantity = 0; if (quantityElement) { quantity = Number( quantityElement.textContent ) || 0; } const customerId = customer.id || customer.mobile || customer.name; saveDeliveryStatus( customerId, "delivered", quantity ); const deliverButton = document.getElementById( "deliverButton" ); if (deliverButton) { deliverButton.textContent = "✓ DELIVERED"; deliverButton.classList.add( "delivered-done" ); deliverButton.disabled = true; } alert( "Milk delivery marked as completed." ); /* Move to next customer */ setTimeout( function() { showNextCustomer(); }, 300 ); } /* ========================================================= SKIP DELIVERY ========================================================= */ function markNotDelivered() { const customers = getCustomers(); if ( customers.length === 0 ) { alert( "No customers available." ); return; } const customer = customers[ currentCustomerIndex ]; if (!customer) { return; } const reason = prompt( "Reason for not delivering:" ); if (reason === null) { return; } const customerId = customer.id || customer.mobile || customer.name; saveDeliveryStatus( customerId, "not-delivered", 0, reason || "Not specified" ); alert( "Delivery skipped.\nReason: " + ( reason || "Not specified" ) ); setTimeout( function() { showNextCustomer(); }, 300 ); } /* ========================================================= SHOW NEXT CUSTOMER ========================================================= */ function showNextCustomer() { const customers = getCustomers(); currentCustomerIndex++; if ( currentCustomerIndex >= customers.length ) { alert( "All customers have been processed for today." ); /* IMPORTANT: We DO NOT remove: milkManagerLoggedIn We DO NOT clear localStorage. We simply stay on the delivery page. */ return; } displayCustomer( customers[ currentCustomerIndex ] ); } /* ========================================================= DISPLAY CUSTOMER ========================================================= */ function displayCustomer( customer ) { if (!customer) { return; } const nameElement = document.getElementById( "customerName" ); const addressElement = document.getElementById( "customerAddress" ); const phoneElement = document.getElementById( "customerPhone" ); const qtyElement = document.getElementById( "milkQty" ); const amountElement = document.getElementById( "milkAmount" ); if (nameElement) { nameElement.textContent = customer.name || "Customer"; } if (addressElement) { addressElement.textContent = customer.address || ""; } if (phoneElement) { phoneElement.textContent = customer.mobile || customer.phone || ""; } const milk = Number( customer.milk || customer.quantity || 2 ); if (qtyElement) { qtyElement.textContent = milk; } if (amountElement) { amountElement.textContent = milk + " L"; } /* Reset buttons */ const deliverButton = document.getElementById( "deliverButton" ); if (deliverButton) { deliverButton.textContent = "✓ DELIVER"; deliverButton.disabled = false; deliverButton.classList.remove( "delivered-done" ); } } /* ========================================================= CHANGE MILK ========================================================= */ function changeMilk( amount ) { const qtyElement = document.getElementById( "milkQty" ); if (!qtyElement) { return; } let quantity = Number( qtyElement.textContent ) || 0; quantity += amount; if (quantity < 0) { quantity = 0; } if (quantity > 20) { quantity = 20; } qtyElement.textContent = quantity; const amountElement = document.getElementById( "milkAmount" ); if (amountElement) { amountElement.textContent = quantity + " L"; } } /* ========================================================= INITIALIZE DELIVERY PAGE ========================================================= */ function initializeDelivery() { const customers = getCustomers(); if ( customers.length === 0 ) { console.log( "No customers found." ); return; } currentCustomerIndex = 0; displayCustomer( customers[0] ); } /* ========================================================= BUTTON CONNECTION ========================================================= */ const deliverButton = document.getElementById( "deliverButton" ); if (deliverButton) { deliverButton.addEventListener( "click", function() { markDelivered(); } ); } const skipButton = document.getElementById( "skipButton" ); if (skipButton) { skipButton.addEventListener( "click", function() { markNotDelivered(); } ); } /* ========================================================= START ========================================================= */ initializeDelivery();
